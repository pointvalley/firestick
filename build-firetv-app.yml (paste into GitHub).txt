name: Build Firestick Home app

on:
  push:
    branches: [main]
    paths:
      - 'firetv-app/**'
      - '.github/workflows/build-firetv-app.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build APK
        working-directory: firetv-app
        env:
          VERSION_CODE: ${{ github.run_number }}
        run: gradle assembleRelease --no-daemon --stacktrace

      - name: Rename APK
        run: cp firetv-app/app/build/outputs/apk/release/app-release.apk firestick-home.apk

      - uses: actions/upload-artifact@v4
        with:
          name: firestick-home
          path: firestick-home.apk

      - name: Publish as latest release
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "app-${{ github.run_number }}" firestick-home.apk \
            --title "Firestick Home app (build ${{ github.run_number }})" \
            --notes "Install on the Firestick with Downloader: pointvalley.github.io/firestick/get" \
            --latest
